add_library(veyra_metrics STATIC metrics.cpp)
target_include_directories(veyra_metrics PUBLIC ${PROJECT_SOURCE_DIR}/include)
target_compile_features(veyra_metrics PUBLIC cxx_std_20)
veyra_enable_warnings(veyra_metrics)

if(BUILD_TESTING)
  enable_testing()
  add_executable(veyra_metrics_tests ${PROJECT_SOURCE_DIR}/tests/test_metrics.cpp)
  target_link_libraries(veyra_metrics_tests PRIVATE veyra_metrics)
  target_compile_features(veyra_metrics_tests PRIVATE cxx_std_20)
  veyra_enable_warnings(veyra_metrics_tests)
  veyra_enable_assertions(veyra_metrics_tests)
  add_test(NAME metrics COMMAND veyra_metrics_tests)
endif()
